TRIGGER|trg_user_events_unread_insert_v268|user_events|CREATE TRIGGER trg_user_events_unread_insert_v268
        AFTER INSERT ON user_events
        WHEN NEW.direction='in'
          AND NEW.event_type IN ('telegram_audio','telegram_document','telegram_message','telegram_photo','telegram_sticker','telegram_video','telegram_voice')
           AND NEW.tg_id NOT IN (53976335)
        BEGIN
            INSERT INTO user_message_state(
                tg_id,last_read_event_id,last_incoming_event_id,unread_count,
                last_message_at,last_message_text
            ) VALUES(
                NEW.tg_id,0,NEW.id,1,NEW.created_at,SUBSTR(COALESCE(NEW.text,''),1,500)
            )
            ON CONFLICT(tg_id) DO UPDATE SET
                last_incoming_event_id=MAX(user_message_state.last_incoming_event_id,NEW.id),
                unread_count=CASE
                    WHEN NEW.id>user_message_state.last_read_event_id
                    THEN MAX(0,user_message_state.unread_count)+1
                    ELSE MAX(0,user_message_state.unread_count)
                END,
                last_message_at=NEW.created_at,
                last_message_text=SUBSTR(COALESCE(NEW.text,''),1,500);
        END
